eight_weeks_ago_start = DATETIME(current_week_start, '-8 weeks')

# Calculate the end of the desired period: the day before the current week's start.
# This ensures the current week's transactions are excluded.
end_of_period = DATETIME(current_week_start, '-1 day')

# Step 2: Filter transactions to include only those within the defined 8-week period.
# For each filtered transaction, calculate its week's start date and determine if it occurred on a weekend.

    (Transactions.date_time >= eight_weeks_ago_ago_start) &
    (Transactions.date_time <= end_of_period)
).CALCULATE(
    # 'transaction_week_start' will be used as the partition key to group transactions by week.
    transaction_week_start=DATETIME(date_time, 'start of week'),
    # 'is_weekend_transaction' is a flag (1 for weekend, 0 for weekday) to easily sum weekend transactions.
    is_weekend_transaction=IFF(
        (DAYNAME(date_time) == "Saturday") | (DAYNAME(date_time) == "Sunday"),
        1,  # Assign 1 if it's a Saturday or Sunday
        0   # Assign 0 otherwise
    )

).CALCULATE(
    week_start=transaction_week_start,  # Include the week start date in the final output
    total_transactions=COUNT(Transactions),  # Count all transactions within each weekly group
