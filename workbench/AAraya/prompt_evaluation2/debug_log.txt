    ref_current_week_start = DATETIME('now', 'start of week')
)

# Step 2: Filter transactions for stocks within the last 8 weeks (excluding current week)
# and calculate necessary helper fields for grouping and aggregation.
# We access Transactions from the context created above, making ref_current_week_start available.
filtered_stock_transactions = initial_context_with_date.Transactions.WHERE(
    (ticker.ticker_type == 'Stock') &  # Filter for stock transactions
    (date_time >= DATETIME(ref_current_week_start, '-8 weeks')) &  # Transactions from 8 weeks ago
    (date_time < ref_current_week_start)  # Transactions up to (but not including) the start of the current week
).CALCULATE(
    # Determine the actual start date of the week for each transaction. This will be our grouping key.
    txn_week_start_for_grouping = DATETIME(date_time, 'start of week'),
    # Flag if the transaction occurred on a weekend (Saturday or Sunday).
    is_txn_on_weekend = ISIN(DAYNAME(date_time), ("Saturday", "Sunday"))
)

# Step 3: Group the filtered transactions by the calculated week start date.
# Then, for each week, count total stock transactions and weekend stock transactions.
result = filtered_stock_transactions.PARTITION(
    name="weekly_transaction_group", by=(txn_week_start_for_grouping)
).CALCULATE(

    output_week_start_date = txn_week_start_for_grouping,
    # Total number of stock transactions in this week.
    output_total_stock_transactions = COUNT(Transactions),
    # Number of stock transactions that occurred on a weekend in this week.
    # IFF(condition, true_value, false_value) is used to sum 1 for each weekend transaction.
    output_weekend_stock_transactions = SUM(IFF(Transactions.is_txn_on_weekend, 1, 0))
).ORDER_BY(output_week_start_date.DESC())  # Order results by week, most recent first.
